Create the FAQ handler. This is a single-turn handler: user asks a question, handler queries `kb_entries` table, returns the answer in the session's language.

```typescript
import { createClient as createServiceClient } from '@supabase/supabase-js'
import { expireSession } from './session-manager'
import type { Session } from './types'

function getServiceClient() {
  return createServiceClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!
  )
}

export async function faqHandler(message: string, session: Session): Promise<string> {
  const lang = session.language
  const supabase = getServiceClient()

  // Search kb_entries for active entries matching the user's question
  // Use ilike on the question field in the user's language
  const questionField = lang === 'bm' ? 'question_bm' : 'question_en'
  const answerField = lang === 'bm' ? 'answer_bm' : 'answer_en'

  // Try keyword match against kb_entries
  const keywords = message.toLowerCase().split(/\s+/).filter(w => w.length > 2)
  let bestMatch: string | null = null

  for (const keyword of keywords) {
    const { data } = await supabase
      .from('kb_entries')
      .select(`${questionField}, ${answerField}`)
      .eq('is_active', true)
      .ilike(questionField, `%${keyword}%`)
      .limit(1)
      .single()

    if (data && data[answerField]) {
      bestMatch = data[answerField] as string
      break
    }
  }

  // If no keyword match, try fetching a general FAQ entry
  if (!bestMatch) {
    const { data: generalFaq } = await supabase
      .from('kb_entries')
      .select(answerField)
      .eq('is_active', true)
      .eq('category', 'general')
      .limit(1)
      .single()

    if (generalFaq && generalFaq[answerField]) {
      bestMatch = generalFaq[answerField] as string
    }
  }

  // Save call record
  const { data: callData } = await supabase.from('calls').insert({
    channel: 'chat',
    wa_number: session.wa_phone,
    language: lang,
    category: 'faq',
    outcome: bestMatch ? 'resolved' : 'abandoned',
    message_count: 1,
    is_test: false,
  }).select('id').single()

  // Save transcript
  if (callData?.id) {
    await supabase.from('transcripts').insert([
      { call_id: callData.id, speaker: 'user', message },
      { call_id: callData.id, speaker: 'bot', message: bestMatch ?? 'No answer found' },
    ])
  }

  // Expire session after single-turn FAQ
  await expireSession(session.id)

  if (bestMatch) {
    return bestMatch
  }

  return lang === 'bm'
    ? 'Maaf, saya tidak menemui jawapan untuk soalan anda. Sila hubungi talian bantuan MyKasih.'
    : 'Sorry, I could not find an answer to your question. Please contact the MyKasih helpline.'
}
```

Key points:
- Queries `kb_entries` with `is_active=true`
- Searches by keyword match in the user's language field
- Returns answer in session language (answer_bm or answer_en)
- Saves call + transcript to Supabase with `channel='chat'`, `category='faq'`
- Single-turn: expires session after answering

Create the merchant lookup handler. This is a 2-step handler:
- Step 0: Ask for postcode or location
- Step 1: Receive input, query merchants, return top 3-5 results

Since `lib/merchant-lookup.ts` uses `createClient` from `lib/supabase/server.ts` (cookie-based), and chatbot API routes may not have cookie context, query the merchants table directly with service role client instead.

```typescript
import { createClient as createServiceClient } from '@supabase/supabase-js'
import { updateSession, expireSession } from './session-manager'
import type { Session } from './types'

function getServiceClient() {
  return createServiceClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!
  )
}

interface MerchantRow {
  chain: string
  outlet_name: string
  address: string
  postcode: string
  state: string
  city: string
}

export async function merchantHandler(message: string, session: Session): Promise<string> {
  const lang = session.language

  switch (session.step) {
    case 0: {
      await updateSession(session.id, { intent: 'merchant_lookup', step: 1 })
      return lang === 'bm'
        ? 'Sila masukkan poskod anda atau nama negeri/bandar (cth: 43000 atau Selangor):'
        : 'Please enter your postcode or state/city name (e.g. 43000 or Selangor):'
    }

    case 1: {
      const input = message.trim()
      const supabase = getServiceClient()
      let merchants: MerchantRow[] = []

      // Detect if input is a postcode (4-5 digits) or text (state/city)
      const isPostcode = /^\d{4,5}$/.test(input)

      if (isPostcode) {
        const prefix = input.substring(0, 4)
        const { data } = await supabase
          .from('merchants')
          .select('chain, outlet_name, address, postcode, state, city')
          .like('postcode', `${prefix}%`)
          .limit(5)
        merchants = (data ?? []) as MerchantRow[]
      } else {
        // Try state match first, then city
        const { data: stateData } = await supabase
          .from('merchants')
          .select('chain, outlet_name, address, postcode, state, city')
          .ilike('state', `%${input}%`)
          .limit(5)

        if (stateData && stateData.length > 0) {
          merchants = stateData as MerchantRow[]
        } else {
          const { data: cityData } = await supabase
            .from('merchants')
            .select('chain, outlet_name, address, postcode, state, city')
            .ilike('city', `%${input}%`)
            .limit(5)
          merchants = (cityData ?? []) as MerchantRow[]
        }
      }

      // Save call record
      const { data: callData } = await supabase.from('calls').insert({
        channel: 'chat',
        wa_number: session.wa_phone,
        language: lang,
        category: 'merchant_lookup',
        outcome: merchants.length > 0 ? 'resolved' : 'abandoned',
        message_count: 2,
        is_test: false,
      }).select('id').single()

      if (callData?.id) {
        await supabase.from('transcripts').insert([
          { call_id: callData.id, speaker: 'user', message },
          { call_id: callData.id, speaker: 'bot', message: `Found ${merchants.length} merchants` },
        ])
      }

      await expireSession(session.id)

      if (merchants.length === 0) {
        return lang === 'bm'
          ? 'Maaf, tiada kedai ditemui di kawasan tersebut. Cuba poskod lain.'
          : 'Sorry, no merchants found in that area. Try a different postcode.'
      }

      const header = lang === 'bm' ? 'Kedai berdekatan:' : 'Nearby merchants:'
      const list = merchants
        .slice(0, 5)
        .map((m, i) => `${i + 1}. ${m.chain} - ${m.outlet_name}\n   ${m.address} (${m.postcode})`)
        .join('\n\n')

      return `${header}\n\n${list}`
    }

    default: {
      await expireSession(session.id)
      return lang === 'bm' ? 'Sesi tamat. Sila mulakan semula.' : 'Session ended. Please start again.'
    }
  }
}
```

Key points:
- 2-step: ask for location (step 0), return results (step 1)
- Detects postcode (4-5 digits) vs text (state/city name)
- Returns top 3-5 formatted merchants
- Saves call with `channel='chat'`, `category='merchant_lookup'`
- Direct Supabase query (not through merchant-lookup.ts which needs cookies)
