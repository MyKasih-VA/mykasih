```typescript
async function getActiveVoiceSessions() {
  const apiKey = process.env.ELEVENLABS_API_KEY
  const agentId = process.env.ELEVENLABS_AGENT_ID
  if (!apiKey || !agentId) return []

  try {
    const res = await fetch(
      `https://api.elevenlabs.io/v1/convai/conversations?agent_id=${agentId}&status=active`,
      {
        headers: { 'xi-api-key': apiKey },
        signal: AbortSignal.timeout(5000),
      }
    )
    if (!res.ok) return []
    const data = await res.json()
    // Map EL conversations to a normalized shape
    return (data.conversations || []).map((conv: Record<string, unknown>) => ({
      id: conv.conversation_id as string,
      caller_name: (conv.metadata as Record<string, unknown>)?.caller_name as string || null,
      wa_number: (conv.metadata as Record<string, unknown>)?.phone as string || null,
      language: (conv.metadata as Record<string, unknown>)?.language as string || null,
      started_at: conv.start_time as string || new Date().toISOString(),
    }))
  } catch {
    return []
  }
}
```

**Chat sessions (D-16):**
```typescript
async function getActiveChatSessions() {
  const supabase = createServiceClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!
  )

  const { data, error } = await supabase
    .from('sessions')
    .select('id, wa_phone, language, intent, step, created_at, expires_at')
    .gt('expires_at', new Date().toISOString())
    .order('created_at', { ascending: false })

  if (error) return []

  return (data || []).map(session => ({
    id: session.id,
    wa_number: session.wa_phone,
    intent: session.intent,
    language: session.language,
    started_at: session.created_at,
    message_count: session.step || 0,
  }))
}
```
