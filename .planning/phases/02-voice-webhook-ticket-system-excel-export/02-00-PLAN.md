  beforeEach(() => {
    jest.clearAllMocks()
  })

  it('returns TKT-2026-00001 when table is empty', async () => {

    mockSingle.mockResolvedValue({ data: { reference_no: 'TKT-2026-00099' }, error: null })
    const ref = await generateTicketRef()
    expect(ref).toBe('TKT-2026-00100')
  })
})
