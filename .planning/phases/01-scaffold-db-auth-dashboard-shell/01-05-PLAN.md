```typescript
import { NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { createClient as createServiceClient } from '@supabase/supabase-js'
import { readFile } from 'fs/promises'
import { join } from 'path'

export async function POST() {
  // Guard 1: Admin role only — validate via Supabase Auth
  const supabase = await createClient()
  const { data: { user }, error: authError } = await supabase.auth.getUser()

  if (authError || !user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const { data: userRecord } = await supabase
    .from('users')
    .select('role')
    .eq('id', user.id)
    .single()

  if (userRecord?.role !== 'admin') {
    return NextResponse.json({ error: 'Forbidden — admin only' }, { status: 403 })
  }

  // Guard 2: Idempotency — if merchants already exist, return 409
  // Use service role client to bypass RLS for count query
  const serviceClient = createServiceClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!
  )

  const { count } = await serviceClient
    .from('merchants')
    .select('*', { count: 'exact', head: true })

  if (count && count > 0) {
    return NextResponse.json(
      { error: 'Already seeded', count },
      { status: 409 }
    )
  }

  // Read merchants.json from project root
  const startTime = Date.now()
  const filePath = join(process.cwd(), '..', 'merchants.json')
  const raw = await readFile(filePath, 'utf-8')
  const merchants = JSON.parse(raw) as Array<{
    chain: string
    outlet_name: string
    state: string
    city: string
    postcode: string
    address: string
  }>

  // Guard 3: Batch insert in chunks of 500
  const CHUNK_SIZE = 500
  let inserted = 0

  for (let i = 0; i < merchants.length; i += CHUNK_SIZE) {
    const chunk = merchants.slice(i, i + CHUNK_SIZE)
    const { error: insertError } = await serviceClient
      .from('merchants')
      .insert(chunk)

    if (insertError) {
      return NextResponse.json(
        { error: `Insert failed at chunk ${Math.floor(i / CHUNK_SIZE)}`, detail: insertError.message },
        { status: 500 }
      )
    }
    inserted += chunk.length
  }

  const durationMs = Date.now() - startTime

  // Guard 4: Return insert summary
  return NextResponse.json({
    inserted,
    skipped: 0,
    duration_ms: durationMs,
  })
}
```

Key details per D-14:
1. Admin role check via Supabase JWT + users table
2. Idempotency: count check returns 409 if >0
3. Batch in chunks of 500 to avoid Supabase payload limits
4. Return `{ inserted, skipped, duration_ms }`

Uses service role client (`SUPABASE_SERVICE_ROLE_KEY`) to bypass RLS for the bulk insert — this is a seed operation. The regular `createClient()` from server.ts is used for auth check only.

NOTE: `merchants.json` is at the project root (`C:/Users/navie/MyKasih/merchants.json`), one level above `mykasih-crm/`. The `join(process.cwd(), '..', 'merchants.json')` path handles this since Next.js cwd is `mykasih-crm/`.

```typescript
import { createClient } from '@/lib/supabase/server'

export interface Merchant {
  id: string
  chain: string
  outlet_name: string
  state: string
  city: string
  postcode: string
  address: string
}

/**
 * Lookup merchants by postcode prefix (first 4 digits).
 * Returns nearest outlets matching the area.
 * MERCH-02
 */
export async function lookupByPostcode(postcode: string): Promise<Merchant[]> {
  const supabase = await createClient()
  const prefix = postcode.trim().substring(0, 4)

  const { data, error } = await supabase
    .from('merchants')
    .select('*')
    .like('postcode', `${prefix}%`)
    .limit(10)

  if (error) {
    console.error('Merchant postcode lookup error:', error)
    return []
  }

  return (data ?? []) as Merchant[]
}

/**
 * Lookup merchants by state and optional city.
 * Returns top 10 matching outlets.
 * MERCH-03
 */
export async function lookupByState(
  state: string,
  city?: string
): Promise<Merchant[]> {
  const supabase = await createClient()

  let query = supabase
    .from('merchants')
    .select('*')
    .ilike('state', state.trim())

  if (city) {
    query = query.ilike('city', city.trim())
  }

  const { data, error } = await query.limit(10)

  if (error) {
    console.error('Merchant state lookup error:', error)
    return []
  }

  return (data ?? []) as Merchant[]
}
```

Key details:
- `lookupByPostcode` matches first 4 digits using LIKE prefix query (per MERCH-02)
- `lookupByState` filters by state (case-insensitive via ilike) + optional city (per MERCH-03)
- Both return max 10 results
- Both use the server client (for server-side usage in API routes)
- Merchant interface matches the Supabase table schema from supabase_migrations.sql
