**app/(dashboard)/page.tsx** — Dashboard home page ('use client'):

```typescript
'use client'

import { useState, useEffect } from 'react'
import { useLanguage } from '@/hooks/useLanguage'
import { t } from '@/lib/translations'
import { StatCard } from '@/components/dashboard/StatCard'
import { CallVolumeChart } from '@/components/dashboard/CallVolumeChart'
import { CategoryDonut } from '@/components/dashboard/CategoryDonut'
import { RecentInteractions } from '@/components/dashboard/RecentInteractions'

interface DashboardData { ... }

export default function DashboardPage() {
  const { language } = useLanguage()
  const [data, setData] = useState<DashboardData | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    fetch('/api/analytics/summary')
      .then(res => res.json())
      .then(setData)
      .catch(console.error)
      .finally(() => setLoading(false))
  }, [])

  return (
    <div>
      {/* Stat cards — 4-column grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
        <StatCard ... />  {/* 4 stat cards */}
      </div>

      {/* Charts — 2fr / 1fr grid */}
      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 mt-4">
        <div className="lg:col-span-2">
          <CallVolumeChart ... />
        </div>
        <CategoryDonut ... />
      </div>

      {/* Recent interactions table */}
      <div className="mt-4">
        <RecentInteractions ... />
      </div>
    </div>
  )
}
```

Grid layout per UI-SPEC:
- Stat cards: `grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4`
- Charts: `grid-cols-1 lg:grid-cols-3 gap-4 mt-4` (bar chart takes col-span-2, donut takes 1)
- Recent table: full width, mt-4

Wire all 4 stat cards with data from API response:
1. Today's Interactions — value: `stats.todayTotal`, subInfo: `${stats.todayVoice} voice / ${stats.todayChat} chat`
2. Resolution Rate — value: `${stats.resolutionRate}%`
3. Open Tickets — value: `stats.openTickets`, subInfo: `${stats.inProgressTickets} in progress`
4. Avg. Duration — value: `${stats.avgDuration}s`, subInfo: `${stats.avgMessages} avg msgs (chat)`

Pass `loading` prop to all components. When `loading=true`, all show skeletons. When `data` is null after loading, show empty states.
