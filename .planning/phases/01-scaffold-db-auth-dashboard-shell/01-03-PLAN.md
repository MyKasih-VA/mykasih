**proxy.ts** — Create at `mykasih-crm/proxy.ts` (NOT middleware.ts per research — Next.js 16 uses proxy.ts):

```typescript
import { NextResponse, type NextRequest } from 'next/server'
import { createServerClient } from '@supabase/ssr'

export async function proxy(request: NextRequest) {
  let supabaseResponse = NextResponse.next({ request })

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll()
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value }) =>
            request.cookies.set(name, value)
          )
          supabaseResponse = NextResponse.next({ request })
          cookiesToSet.forEach(({ name, value, options }) =>
            supabaseResponse.cookies.set(name, value, options)
          )
        },
      },
    }
  )

  const { data: { user } } = await supabase.auth.getUser()

  const pathname = request.nextUrl.pathname

  // Protected routes — all dashboard pages
  const protectedPaths = [
    '/',
    '/voice-calls',
    '/chat-messages',
    '/all-interactions',
    '/tickets',
    '/beneficiaries',
    '/live-monitor',
    '/analytics',
    '/knowledge-base',
    '/staff',
    '/integrations',
    '/testing',
    '/settings',
  ]

  const isProtected = protectedPaths.some(p =>
    pathname === p || pathname.startsWith(p + '/')
  )

  if (!user && isProtected) {
    return NextResponse.redirect(new URL('/login', request.url))
  }

  if (user && pathname === '/login') {
    return NextResponse.redirect(new URL('/', request.url))
  }

  return supabaseResponse
}

export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|jpeg|gif|webp)$).*)',
  ],
}
```

Export function name is `proxy` (NOT `middleware`). The config.matcher excludes static assets. Uses `getAll`/`setAll` cookie pattern inline (cannot import from lib in proxy file). Uses `getUser()` not `getSession()` for security (per Supabase docs — getSession reads from cookies which can be tampered).

**hooks/useLanguage.ts** — Language toggle hook with localStorage AND Supabase users.language sync (per D-04 / DASH-04):

```typescript
'use client'

import { useState, useEffect, useCallback } from 'react'
import { type Language } from '@/lib/translations'
import { createClient } from '@/lib/supabase/client'

const STORAGE_KEY = 'mykasih-language'
const DEFAULT_LANGUAGE: Language = 'en'

export function useLanguage() {
  const [language, setLanguageState] = useState<Language>(DEFAULT_LANGUAGE)
  const [isLoaded, setIsLoaded] = useState(false)

  useEffect(() => {
    const stored = localStorage.getItem(STORAGE_KEY) as Language | null
    if (stored === 'en' || stored === 'bm') {
      setLanguageState(stored)
    }
    setIsLoaded(true)
  }, [])

  const setLanguage = useCallback(async (lang: Language) => {
    setLanguageState(lang)
    localStorage.setItem(STORAGE_KEY, lang)

    // Per DASH-04: sync language preference to users.language in Supabase
    try {
      const supabase = createClient()
      const { data: { user } } = await supabase.auth.getUser()
      if (user) {
        await supabase
          .from('users')
          .update({ language: lang })
          .eq('id', user.id)
      }
    } catch {
      // Supabase sync is best-effort — localStorage is the primary source
      // If user is not authenticated (e.g. login page), this silently fails
    }
  }, [])

  const toggleLanguage = useCallback(() => {
    setLanguage(language === 'en' ? 'bm' : 'en')
  }, [language, setLanguage])

  return { language, setLanguage, toggleLanguage, isLoaded }
}
```

The hook reads from localStorage on mount, defaults to 'en', and provides both setLanguage and toggleLanguage. The `isLoaded` flag prevents flash of wrong language on first render. Per DASH-04, every language change also updates the `users.language` field in Supabase via the browser client. The Supabase sync is best-effort (try/catch) so the UI is never blocked by a network call.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser -> proxy.ts | All requests pass through proxy; unauthenticated users blocked from protected routes |
| Login form -> Supabase Auth | Email/password sent to Supabase Auth API (HTTPS) |
| proxy.ts -> Supabase Auth | Server-side user verification via getUser() |
| useLanguage -> Supabase users | Language preference update via authenticated client |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Spoofing | Login form | mitigate | Uses Supabase Auth (server-side verification via getUser(), not client-side getSession()); bcrypt password hashing handled by Supabase |
| T-03-02 | Tampering | JWT cookies | mitigate | proxy.ts calls getUser() which validates JWT server-side against Supabase; tampered cookies return null user -> redirect to login |
| T-03-03 | Information Disclosure | Error messages | mitigate | Generic "Invalid email or password" message — does not reveal whether email exists (Supabase default behavior) |
| T-03-04 | Denial of Service | Login attempts | accept | Rate limiting handled by Supabase Auth (built-in); not custom-implementing at application layer for v1 |
| T-03-05 | Elevation of Privilege | Role redirect | mitigate | Role read from `users` table after auth (server can verify); proxy.ts checks auth on every protected route — role only affects initial redirect destination, not access control (RLS handles data access) |
| T-03-06 | Tampering | users.language update | mitigate | Update uses authenticated client (getUser() check); RLS on users table restricts updates to own row only |
</threat_model>
