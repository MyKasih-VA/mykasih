```typescript
async function handleStatusChange(ticketId: string, newStatus: TicketStatus) {
  // 1. Save old state for revert
  const prevTickets = [...tickets]
  // 2. Optimistic update — move card immediately
  setTickets(prev => prev.map(t => t.id === ticketId ? { ...t, status: newStatus } : t))
  setUpdateError(null)
  // 3. PATCH in background
  try {
    const res = await fetch(`/api/tickets/${ticketId}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ status: newStatus }),
    })
    if (!res.ok) throw new Error()
  } catch {
    // 4. Revert on failure
    setTickets(prevTickets)
    setUpdateError(t('error.statusUpdate', language))
    setTimeout(() => setUpdateError(null), 3000)
  }
}
```
