```typescript
import { NextResponse, type NextRequest } from "next/server"

export function proxy(request: NextRequest) {
  // Check for Supabase session cookie (format: sb-<project-ref>-auth-token)
  const hasSession = request.cookies.getAll().some(
    (cookie) => cookie.name.startsWith("sb-") && cookie.name.endsWith("-auth-token")
  )

  if (!hasSession) {
    const url = request.nextUrl.clone()
    url.pathname = "/login"
    return NextResponse.redirect(url)
  }

  return NextResponse.next()
}

  matcher: [
    "/((?!_next/static|_next/image|favicon\\.ico|login|api/|demo).*)",
```

Modify `mykasih-crm/proxy.ts` to enforce AAL2 for admin sessions. Without this, an admin who has completed email/password login (aal1) but not MFA challenge can navigate directly to `/dashboard` and bypass the MFA flow.

**Current behavior:** proxy.ts only checks for cookie presence. If a session cookie exists, the request passes through.

**Required behavior:** After confirming a session cookie exists, check if the user is an admin with aal1. If so, redirect to `/login/mfa-challenge`.

**Implementation approach:**

Since proxy.ts (Next.js 16 proxy) runs on the Node.js runtime, we can use the Supabase server client to decode the session. However, proxy.ts should remain lightweight. Use this approach:

1. Parse the Supabase auth token cookie value to extract the JWT payload (base64 decode the middle segment -- no crypto verification needed since Supabase verifies server-side)
2. Check the `aal` claim in the JWT. Supabase JWTs include `aal: 'aal1' | 'aal2'` in the payload.
3. Check the app_metadata or use a lightweight lookup for the user's role. Since the JWT does not contain the app role (it's in the users table, not the JWT), use a simpler heuristic:
   - If `aal` claim is `'aal1'` AND the request path is NOT `/login` or `/login/mfa-*`, redirect to `/login/mfa-challenge`
   - This applies to ALL users with aal1 when MFA factors exist, but since only admins are forced to enroll MFA (via the login page gate), non-admin users will never have `nextLevel === 'aal2'` and their JWT will not trigger this

**Alternative simpler approach (preferred):** Read the Supabase session cookie, decode the JWT payload, and check:
- If `aal` is present and equals `'aal1'` AND the JWT's `amr` array contains a factor with `type: 'totp'` (indicating MFA is enrolled but not verified this session), redirect to `/login/mfa-challenge`

```typescript
// Add inside the proxy function, after hasSession check passes:

// Decode JWT payload from session cookie (no verification -- Supabase handles that)
const sessionCookie = request.cookies.getAll().find(
  (cookie) => cookie.name.startsWith("sb-") && cookie.name.endsWith("-auth-token")
)
if (sessionCookie) {
  try {
    const payload = JSON.parse(
      Buffer.from(sessionCookie.value.split('.')[1], 'base64').toString()
    )
    // If session is aal1 but user has TOTP factors enrolled, force MFA challenge
    // This catches admin users who bypass the login MFA gate via direct URL
    if (payload.aal === 'aal1') {
      const amrMethods = payload.amr?.map((a: { method: string }) => a.method) ?? []
      // If amr includes 'totp' as a registered method but aal is still 1,
      // the user has MFA enrolled but hasn't completed the challenge
      // However, amr only shows methods USED in this session, not enrolled methods
      // So we need a different approach: just check if aal is aal1 and the
      // session metadata indicates MFA should be required
    }
  } catch {
    // Cookie parse failed -- let request through, Supabase will handle auth
  }
}
```
